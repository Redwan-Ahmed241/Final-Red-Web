=== SNIPPET 1: JAVASCRIPT STATE TRACKER (PREFIX: examjs) ===

let total = 0, count = 0;
let items = ""; // String tracking (no array needed!)
// Or using array: const list = [];

function add() {
    let name = document.getElementById("a").value;
    let val = Number(document.getElementById("b").value);
    if (!name || isNaN(val) || val <= 0) return alert("Invalid input");

    count++;
    total += val;
    items += name + ", "; // Just like count++! (Or: list.push(name);)

    let avg = total / count;
    let msg = "";

    if (total <= 5)       msg = "Start studying regularly!";
    else if (total <= 12) msg = "Good progress!";
    else if (total <= 19) msg = "Almost ready!";
    else                  msg = "Goal reached!";

    if (count > 7 && total < 20) msg = "Increase study time!";

    // Simple line-by-line output:
    document.getElementById("res").innerHTML = 
        "Total: " + total + "<br>" +
        "Average: " + avg.toFixed(2) + "<br>" +
        "Subjects: [" + items.slice(0, -2) + "]<br>" +
        "Status: " + msg;

    document.getElementById("a").value = "";
    document.getElementById("b").value = "";
}

=== SNIPPET 4: PHP CEIL ALLOCATION (PREFIX: examphpcalc) ===

<?php
$demand   = floatval($_POST['a']);
$rate     = floatval($_POST['b']);
$capacity = floatval($_POST['c']);

$unitsNeeded = ceil($demand / $capacity);
$totalCap    = $unitsNeeded * $capacity;
$empty       = $totalCap - $demand;
$wastedCost  = $empty * $rate;

$unusedPct = ($totalCap > 0) ? ($empty / $totalCap) * 100 : 0;
$status = ($unusedPct <= 10) ? "Efficient" : (($unusedPct <= 25) ? "Acceptable" : "Wasteful");

echo "Units Needed: " . $unitsNeeded . "<br>";
echo "Empty Slots: " . $empty . "<br>";
echo "Wasted Cost: " . $wastedCost . " BDT<br>";
echo "Status: " . $status;
?>


=== SNIPPET 5: PHP TARGET COMPARISON (PREFIX: examphptarget) ===

<?php
$val1   = floatval($_POST['cred']);
$val2   = floatval($_POST['num_course']);
$target = floatval($_POST['tar_cred']);

$total = $val1 * $val2;

// 4-Tier Categorization
$cat = "Poor";
if ($total >= 120)     $cat = "Excellent";
elseif ($total >= 90)  $cat = "Good";
elseif ($total >= 60)  $cat = "Average";

// Target Comparison (Keep subtraction in parentheses!)
if ($total == $target) {
    $state = "Exactly at target (0 diff)";
} elseif ($total > $target) {
    $state = "Above target by " . ($total - $target);
} else {
    $state = "Below target by " . ($target - $total);
}

echo "Total credits: " . $total . "<br>";
echo "Progress category: " . $cat . "<br>";
echo "Target comparison: " . $state . "<br>";
?>

=== SNIPPET 8: PHP MYSQL RUNNER WITH WHILE LOOP (PREFIX: examphpmysql) ===

<?php
$conn = new mysqli("localhost", "root", "", "exam_db");

// Task 1: Category Summary with while loop
echo "<h3>Task 1: Summary</h3>";
$res1 = $conn->query("SELECT category, COUNT(*) AS total, AVG(price) AS avg_val FROM items GROUP BY category");
while ($row = $res1->fetch_assoc()) {
    echo $row['category'] . " - Total: " . $row['total'] . " - Avg: " . $row['avg_val'] . "<br>";
}

// Task 2: Conditional UPDATE
$conn->query("UPDATE items SET status = 'Discount' WHERE price < 1000 AND status <> 'Pending'");
echo "Task 2 Updated!<br>";

// Task 3: Bonus / Increment with Ceiling Cap
$conn->query("UPDATE items SET price = price + 500 WHERE price > 300 AND (price + 500) <= 60000");
echo "Task 3 Updated!<br>";

// Task 4: Subquery / CASE WHEN with while loop
echo "<h3>Task 4: High Value Items</h3>";
$res4 = $conn->query("SELECT name, price FROM items WHERE price > (SELECT AVG(price) FROM items)");
while ($row = $res4->fetch_assoc()) {
    echo $row['name'] . ": " . $row['price'] . "<br>";
}

// Total Net Worth (Single Value)
$res5 = $conn->query("SELECT SUM(price * stock) AS net_worth FROM items");
$row5 = $res5->fetch_assoc();
echo "<b>Total: " . $row5['net_worth'] . "</b>";

$conn->close();
?>
