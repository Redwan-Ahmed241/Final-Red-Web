=== SNIPPET 1: JAVASCRIPT STATE TRACKER (PREFIX: examjs) ===

const GOAL = 2000;
let total = 0;
let count = 0;
const items = [];

const inputName = document.getElementById("itemName");
const inputValue = document.getElementById("itemValue");
const actionBtn = document.getElementById("actionBtn");
const outputLabel = document.getElementById("outputLabel");

function processEntry() {
    const name = inputName ? inputName.value.trim() : "";
    const val = parseFloat(inputValue.value);

    if (isNaN(val) || val <= 0) {
        alert("Please enter a valid positive number.");
        return;
    }

    count++;
    total += val;
    if (name) items.push(name);

    const average = total / count;
    let feedback = "";

    if (total <= 800) {
        feedback = "Initial progress, keep going!";
    } else if (total <= 1600) {
        feedback = "Good progress, staying balanced!";
    } else if (total < GOAL) {
        feedback = "Almost at the target!";
    } else {
        feedback = "Goal reached successfully!";
    }

    if (count > 8 && total < GOAL) {
        feedback += " Warning: Too many entries without reaching goal!";
    }

    outputLabel.textContent = `Total: ${total} | Average: ${parseFloat(average.toFixed(2))} | Entries: ${count} | Feedback: ${feedback}`;

    if (inputName) inputName.value = "";
    inputValue.value = "";
    inputValue.focus();
}

actionBtn.addEventListener("click", processEntry);


=== SNIPPET 2: JAVASCRIPT FORMULA & ARRAYS (PREFIX: examjsformula) ===

const readingsA = [];
const readingsB = [];

const inputA = document.getElementById("inputA");
const inputB = document.getElementById("inputB");
const calculateBtn = document.getElementById("calculateBtn");
const resultLabel = document.getElementById("resultLabel");

function evaluateFormula() {
    const valA = parseFloat(inputA.value);
    const valB = parseFloat(inputB.value);

    if (isNaN(valA) || isNaN(valB)) {
        alert("Invalid input");
        return;
    }

    readingsA.push(valA);
    readingsB.push(valB);

    const avgA = readingsA.reduce((sum, n) => sum + n, 0) / readingsA.length;
    const avgB = readingsB.reduce((sum, n) => sum + n, 0) / readingsB.length;

    const score = (100 - valB) * 2 + Math.abs(valA - 80) * 0.5;

    let status = "";
    if (score <= 10) {
        status = "SAFE";
    } else if (score <= 20) {
        status = "WARNING";
    } else {
        status = "DANGER";
    }

    resultLabel.textContent = `Average A: ${avgA.toFixed(2)} | Average B: ${avgB.toFixed(2)} | Score: ${score.toFixed(1)} | Status: ${status}`;
    inputA.value = "";
    inputB.value = "";
}

calculateBtn.addEventListener("click", evaluateFormula);

=== SNIPPET 4: PHP CEIL, UNUSED & CLASSIFICATION (PREFIX: examphpcalc) ===

<?php
function computeLogistics($demand, $rate, $capacity) {
    $unitsNeeded = ceil($demand / $capacity);
    $totalCapacity = $unitsNeeded * $capacity;
    $unused = $totalCapacity - $demand;
    $wastedCost = $unused * $rate;
    $unusedPercentage = ($totalCapacity > 0) ? ($unused / $totalCapacity) * 100 : 0;

    $classification = "";
    if ($unusedPercentage <= 10) {
        $classification = "Efficient";
    } elseif ($unusedPercentage <= 25) {
        $classification = "Acceptable";
    } else {
        $classification = "Wasteful";
    }

    return [
        'units' => $unitsNeeded,
        'unused' => $unused,
        'wastedCost' => $wastedCost,
        'classification' => $classification
    ];
}

if ($_SERVER["REQUEST_METHOD"] == "POST") {
    $val1 = floatval($_POST['val1']);
    $val2 = floatval($_POST['val2']);
    $val3 = floatval($_POST['val3']);

    $res = computeLogistics($val1, $val2, $val3);

    echo "<h2>Calculation Result</h2>";
    echo "<table border='1' cellpadding='8'>";
    echo "<tr><th>Demand</th><th>Rate</th><th>Capacity</th><th>Units Needed</th><th>Unused</th><th>Wasted Cost</th><th>Status</th></tr>";
    echo "<tr>
            <td>{$val1}</td>
            <td>{$val2}</td>
            <td>{$val3}</td>
            <td>{$res['units']}</td>
            <td>{$res['unused']}</td>
            <td>" . number_format($res['wastedCost'], 2) . "</td>
            <td>{$res['classification']}</td>
          </tr>";
    echo "</table>";
    echo "<br><a href='index.html'>Back</a>";
}
?>


=== SNIPPET 5: PHP TARGET PERFORMANCE COMPARISON (PREFIX: examphptarget) ===

<?php
function evaluateTargetPerformance($ratePerDay, $days, $target) {
    $total = $ratePerDay * $days;

    $category = "";
    if ($total >= 500) {
        $category = "Excellent";
    } elseif ($total >= 300) {
        $category = "Good";
    } elseif ($total >= 150) {
        $category = "Average";
    } else {
        $category = "Poor";
    }

    $diff = $total - $target;
    $result = "";
    if ($diff == 0) {
        $result = "Target met exactly (0 difference)";
    } elseif ($diff > 0) {
        $result = "Above target by " . $diff;
    } else {
        $result = "Below target by " . abs($diff);
    }

    return ['total' => $total, 'category' => $category, 'result' => $result];
}

if ($_SERVER["REQUEST_METHOD"] == "POST") {
    $r = floatval($_POST['val1']);
    $d = floatval($_POST['val2']);
    $t = floatval($_POST['val3']);

    $data = evaluateTargetPerformance($r, $d, $t);

    echo "<table border='1' cellpadding='8'>";
    echo "<tr><th>Total Produced</th><th>Category</th><th>Result</th></tr>";
    echo "<tr><td>{$data['total']}</td><td>{$data['category']}</td><td>{$data['result']}</td></tr>";
    echo "</table>";
}
?>

=== SNIPPET 7: COMPLETE PHP MYSQL RUNNER (PREFIX: examphpmysql) ===

<?php
$conn = new mysqli("localhost", "root", "", "exam_db");
if ($conn->connect_error) {
    die("Connection failed: " . $conn->connect_error);
}

function displayTable($conn, $title, $sql) {
    echo "<h3>" . htmlspecialchars($title) . "</h3>";
    $result = $conn->query($sql);
    if (!$result) {
        echo "Error: " . $conn->error . "<br>";
        return;
    }
    if ($result->num_rows > 0) {
        echo "<table border='1' cellpadding='6'><tr>";
        while ($field = $result->fetch_field()) {
            echo "<th>" . htmlspecialchars($field->name) . "</th>";
        }
        echo "</tr>";
        while ($row = $result->fetch_assoc()) {
            echo "<tr>";
            foreach ($row as $val) {
                echo "<td>" . htmlspecialchars($val) . "</td>";
            }
            echo "</tr>";
        }
        echo "</table>";
    } else {
        echo "0 results found.<br>";
    }
}

function executeUpdate($conn, $title, $sql) {
    echo "<h3>" . htmlspecialchars($title) . "</h3>";
    if ($conn->query($sql) === TRUE) {
        echo "Updated successfully. Affected rows: " . $conn->affected_rows . "<br>";
    } else {
        echo "Update error: " . $conn->error . "<br>";
    }
}

displayTable($conn, "Task 1: Category Summary", 
    "SELECT category, COUNT(*) AS total_items, AVG(rating) AS avg_rating, SUM(price * stock) AS category_value FROM records GROUP BY category HAVING avg_rating > 4.0 ORDER BY avg_rating DESC");

executeUpdate($conn, "Task 2: Conditional Status Update", 
    "UPDATE records SET category = 'Discounted' WHERE price < 300 AND rating < 4.0");

executeUpdate($conn, "Task 3: Percentage Increment with Upper Ceiling", 
    "UPDATE records SET price = price * 1.10 WHERE price > 400 AND (price * 1.10) <= 1000.00");

displayTable($conn, "Task 4: Item Classification Against Category Average", 
    "SELECT t1.title, t1.category, t1.price, CASE WHEN t1.price > (SELECT AVG(t2.price) FROM records t2 WHERE t2.category = t1.category) THEN 'Premium Tier' ELSE 'Standard Tier' END AS item_tier FROM records t1 ORDER BY t1.price DESC");

$singleResult = $conn->query("SELECT SUM(price * stock) AS total_net_worth FROM records");
if ($singleResult && $row = $singleResult->fetch_assoc()) {
    echo "<p><b>Total Value: " . number_format($row['total_net_worth'], 2) . "</b></p>";
}

$conn->close();
?>
