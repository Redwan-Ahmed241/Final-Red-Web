let total = 0, count = 0;
let items = ""; // String accumulator (Just like count++ without array!)
// Or array way: const list = [];

function add() {
  let name = document.getElementById("a").value;
  let val = Number(document.getElementById("b").value);
  if (!name || isNaN(val) || val <= 0) return alert("Invalid");

  count++;
  total += val;
  items += name + ", "; // Adds to string! (Or: list.push(name);)

  let avg = total / count;
  let msg = "";
  if (total <= 5)       msg = "Start studying regularly!";
  else if (total <= 12) msg = "Good progress!";
  else if (total <= 19) msg = "Almost ready!";
  else                  msg = "Weekly goal reached!";

  if (count > 7 && total < 20) msg = "Increase study time / frequent snacking!";

  // Simple line-by-line output:
  document.getElementById("res").innerHTML =
    "Total: " + total + "<br>" +
    "Average: " + avg.toFixed(2) + "<br>" +
    "Subjects: [" + items.slice(0, -2) + "]<br>" + // Array way: list.join(", ")
    "Status: " + msg;

  document.getElementById("a").value = "";
  document.getElementById("b").value = "";
}
TYPE 1B: CRITERIA & POINTS SCORER (Password Strength) TYPE 1B 
let attempts = 0;
function checkScore() {
  let p = document.getElementById("a").value;
  attempts++;
  let score = 0;
  if (p.length >= 6) score += Math.floor((p.length - 6) / 2) * 10;
  if (/[A-Z]/.test(p))      score += 15; // Uppercase
  if (/[a-z]/.test(p))      score += 15; // Lowercase
  if (/[0-9]/.test(p))      score += 20; // Number
  if (/[!@#$%^&*]/.test(p)) score += 25; // Special char

  let lvl = (score <= 30) ? "Very Weak" : (score <= 50) ? "Weak" :
            (score <= 70) ? "Medium" : (score <= 90) ? "Strong" : "Very Strong";
  let msg = (score >= 100) ? "Perfect Password! Success!" : "Score: " + score + " - " + lvl;
  if (attempts > 8 && score < 71) msg += " | Need practice! Add uppercase, digits & symbols.";
  document.getElementById("res").innerHTML = msg;
}
TYPE 1C: MULTI-INPUT SENSOR & FORMULA (Patient Vitals) TYPE 1C 
const listA = [], listB = [];
function calc() {
  let vA = Number(document.getElementById("a").value);
  let vB = Number(document.getElementById("b").value);
  if (isNaN(vA) || isNaN(vB)) return;
  listA.push(vA); listB.push(vB);

  let avgA = listA.reduce((s, n) => s + n, 0) / listA.length;
  let avgB = listB.reduce((s, n) => s + n, 0) / listB.length;
  let score = (100 - vB) * 2 + Math.abs(vA - 80) * 0.5;
  let status = (score <= 10) ? "SAFE" : (score <= 20) ? "WARNING" : "DANGER";

  document.getElementById("res").innerHTML =
    "Avg A: " + avgA.toFixed(1) + " | Avg B: " + avgB.toFixed(1) + " | Score: " + score.toFixed(1) + " | Status: " + status;
}

ZERO-CRASH EMERGENCY RULES & GOTCHAS
  • WHERE vs HAVING: WHERE filters rows; HAVING filters groups. Never put COUNT(*) or AVG() in WHERE!
  • SQL Order: WHERE > GROUP BY > HAVING > ORDER BY.
  • Execution Order: Run tasks in exact sequence. Early updates affect later queries.
  • SQL Not Equal: Use <> or !=. String in single quotes 'Text'.
  • XAMPP Run: Always run via http://localhost/folder/index.html (never double click file!).
