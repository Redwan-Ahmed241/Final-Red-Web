TYPE 1A: RUNNING TOTAL & GOAL PROGRESS TRACKER (Calorie, Study, Expense) TYPE 1A 
const GOAL = 2000; // Target from exam (e.g. 2000 kcal, 20 hrs, 100 pts)
let total = 0, count = 0;
const historyList = []; // Stores entered subjects/meals

function handleAction() {
  const val = parseFloat(document.getElementById("inputB").value);
  const name = document.getElementById("inputA").value.trim();
  if (isNaN(val) || val <= 0) { alert("Enter valid positive number"); return; }
  count++; total += val;
  if (name) historyList.push(name);
  const avg = total / count;

  // 4-Tier Ladder (Order: smallest first with <= OR largest first with >=)
  let msg = "";
  if (total <= 800)       msg = "You’re off to a healthy start!";
  else if (total <= 1600) msg = "Good progress, keep it balanced!";
  else if (total < GOAL)  msg = "Almost at your limit!";
  else                    msg = "Goal reached! Stay mindful!";

  // Exam Warning Limit: Too many entries/sessions without reaching goal
  if (count > 10 && total < GOAL) msg += " Be cautious of frequent snacking!";

  document.getElementById("outLabel").textContent =
    `Total: ${total} | Avg: ${parseFloat(avg.toFixed(2))} | Items: [${historyList.join(", ")}] | ${msg}`;
  document.getElementById("inputB").value = "";
}
document.getElementById("actionBtn").addEventListener("click", handleAction);
TYPE 1B: CRITERIA & POINTS SCORER (Password Strength, Scoring) TYPE 1B 
let attempts = 0;
function checkScore() {
  const p = document.getElementById("inputA").value;
  attempts++;
  let score = 0;
  // Criteria 1: Length points (e.g. +10 per 2 chars above min 6 chars)
  if (p.length >= 6) score += Math.floor((p.length - 6) / 2) * 10;
  // Criteria 2-5: Regex tests for character sets
  if (/[A-Z]/.test(p))      score += 15; // Uppercase
  if (/[a-z]/.test(p))      score += 15; // Lowercase
  if (/[0-9]/.test(p))      score += 20; // Number
  if (/[!@#$%^&*]/.test(p)) score += 25; // Special char

  // Tiers: Very Weak (0-30), Weak (31-50), Medium (51-70), Strong (71-90), Very Strong (91+)
  let lvl = (score <= 30) ? "Very Weak" : (score <= 50) ? "Weak" :
            (score <= 70) ? "Medium" : (score <= 90) ? "Strong" : "Very Strong";
  let out = (score >= 100) ? "Perfect Password! Success!" : `Score: ${score} - ${lvl}`;

  // Warning on > N attempts without reaching "Strong" (score < 71)
  if (attempts > 8 && score < 71) out += " | Need practice! Tips: Add uppercase, digits & symbols.";
  document.getElementById("outLabel").textContent = out;
}
TYPE 1C: MULTI-INPUT SENSOR & FORMULA EVALUATOR (Patient Vitals) TYPE 1C 
const listA = [], listB = []; // Keeps track of all readings
function computeFormula() {
  const vA = parseFloat(document.getElementById("inputA").value);
  const vB = parseFloat(document.getElementById("inputB").value);
  if (isNaN(vA) || isNaN(vB)) return;
  listA.push(vA); listB.push(vB);

  // Averages using reduce
  const avgA = listA.reduce((s, n) => s + n, 0) / listA.length;
  const avgB = listB.reduce((s, n) => s + n, 0) / listB.length;

  // Formula with absolute difference: Math.abs(vA - 80)
  const score = (100 - vB) * 2 + Math.abs(vA - 80) * 0.5;
  const status = (score <= 10) ? "SAFE" : (score <= 20) ? "WARNING" : "DANGER";

  document.getElementById("outLabel").textContent =
    `Avg A: ${avgA.toFixed(1)} | Avg B: ${avgB.toFixed(1)} | Score: ${score.toFixed(1)} | Status: ${status}`;
}

function calculateBooking($people, $costPerPerson, $capacity) {
  // 1. Cannot book partial units: always round UP with ceil()
  $unitsNeeded = ceil($people / $capacity);
  // 2. Empty / Unused capacity
  $totalCapacity = $unitsNeeded * $capacity;
  $emptyUnits = $totalCapacity - $people;
  // 3. Wasted money: empty seats * ticket/person price
  $wastedMoney = $emptyUnits * $costPerPerson;

  // 4. Efficiency status (% unused of total carried)
  $unusedPct = ($totalCapacity > 0) ? ($emptyUnits / $totalCapacity) * 100 : 0;
  $status = ($unusedPct <= 10) ? "Efficient" : (($unusedPct <= 25) ? "Acceptable" : "Wasteful");

  return ['units' => $unitsNeeded, 'empty' => $emptyUnits, 'waste' => $wastedMoney, 'status' => $status];
}

function evaluatePerformance($rate, $duration, $target) {

ALL 6 SQL QUERY ARCHETYPES (COVERING 100% OF PAST & FUTURE EXAMS) SQL PATTERNS 
-- ARCHETYPE 1: Category Summary & Aggregation with HAVING
SELECT category, COUNT(*) AS total_items, AVG(rating) AS avg_rating, SUM(price) AS total_val
FROM records GROUP BY category HAVING AVG(rating) > 4.0 ORDER BY avg_rating DESC;

-- ARCHETYPE 2: Conditional Filtered SELECT with OR / LIKE / BETWEEN
SELECT name, category, rating FROM records
WHERE rating > 4.5 OR name LIKE '%Beach%' ORDER BY rating DESC;

-- ARCHETYPE 3: Multi-Column Conditional UPDATE
UPDATE records SET status = 'Grace Period', price = 0.00
WHERE status = 'Overdue' AND stock < 7;

-- ARCHETYPE 4: Capped Increments & Percentage Discounts (+10%, -10%, +5000 with CAP)
-- 4a. Percentage increase (+10%):
UPDATE records SET price = price * 1.10 WHERE price > 400;
-- 4b. Percentage discount (-10%):
UPDATE records SET price = price * 0.90 WHERE category = 'Language';
-- 4c. Fixed bonus with upper ceiling cap:
UPDATE records SET price = price + 5000 WHERE price > 50000 AND (price + 5000) <= 60000;

-- ARCHETYPE 5: Subquery & Category Average Comparison (CASE WHEN ... END)
SELECT r1.name, r1.category, r1.price,
  CASE WHEN r1.price > (SELECT AVG(r2.price) FROM records r2 WHERE r2.category = r1.category)
       THEN 'Top Seller' ELSE 'Regular Seller' END AS seller_status
FROM records r1 ORDER BY r1.price DESC;

-- ARCHETYPE 6: Whole Table Single Value (Total Net Worth: SUM(price * stock))
SELECT SUM(price * stock) AS total_net_worth FROM records;
