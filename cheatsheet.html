<?php
// 1. Database Connection
$conn = new mysqli("localhost", "root", "", "uiu_exam");
if ($conn->connect_error) {
  die("Connection failed: " . $conn->connect_error);
}

// 2. Reusable Helper: Executes SELECT and displays an HTML Table
function showQuery($conn, $title, $sql) {
  echo "<h4>$title</h4>";
  $result = $conn->query($sql);
  if (!$result) { echo "Error: " . $conn->error; return; }
  if ($result->num_rows > 0) {
    echo "<table border='1' cellpadding='5'><tr>";
    while ($f = $result->fetch_field()) echo "<th>{$f->name}</th>";
    echo "</tr>";
    while ($row = $result->fetch_assoc()) {
      echo "<tr>";
      foreach ($row as $val) echo "<td>$val</td>";
      echo "</tr>";
    }
    echo "</table>";
  } else { echo "0 results found.<br>"; }
}

// 3. Reusable Helper: Executes UPDATE / DELETE queries
function executeUpdate($conn, $title, $sql) {
  echo "<h4>$title</h4>";
  if ($conn->query($sql) === TRUE) {
    echo "Updated successfully. Rows affected: " . $conn->affected_rows . "<br>";
  } else { echo "Update error: " . $conn->error . "<br>"; }
}

// Task 1: Category Aggregation
$sql1 = "SELECT category, COUNT(*) AS total_items FROM records GROUP BY category";
showQuery($conn, "Task 1: Category Count", $sql1);

// Task 2: Conditional Update with WHERE ... AND ...
$sql2 = "UPDATE records SET status = 'Grace Period' WHERE status = 'Overdue' AND stock < 10";
executeUpdate($conn, "Task 2: Status Update", $sql2);

// Task 3: Bonus / Percentage Update with Upper Ceiling Cap
$sql3 = "UPDATE records SET price = price * 1.10 WHERE price > 400 AND (price * 1.10) <= 1000";
executeUpdate($conn, "Task 3: Price +10% with Cap", $sql3);

// Task 4: Advanced SELECT with Sorting or Category Average Subquery
